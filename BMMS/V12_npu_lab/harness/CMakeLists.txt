set(NPU_ARCH "dav-2201" CACHE STRING "Architecture used by the supplied Judge CMake")

foreach(version IN LISTS BMMS_VERSIONS)
  add_executable(bench_${version} main.asc)
  target_compile_definitions(bench_${version} PRIVATE BMMS_KERNEL_HEADER="${version}.asc" BMMS_VARIANT="${version}")
  target_link_libraries(bench_${version} PRIVATE tiling_api register platform unified_dlog dl m graph_base)
  target_include_directories(bench_${version} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_options(bench_${version} PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=${NPU_ARCH}>)
endforeach()
