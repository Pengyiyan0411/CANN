cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(bmms_small LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 17)
foreach(v r68 r69 r70 r71 r72)
 add_executable(bench_${v} main.asc)
 target_compile_definitions(bench_${v} PRIVATE BMMS_KERNEL_HEADER="${v}.asc" BMMS_VARIANT="${v}")
 target_link_libraries(bench_${v} PRIVATE tiling_api register platform unified_dlog dl m graph_base)
 target_compile_options(bench_${v} PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=dav-2201>)
endforeach()
add_executable(sanitize_r69 main.asc)
target_compile_definitions(sanitize_r69 PRIVATE BMMS_KERNEL_HEADER="r69.asc" BMMS_VARIANT="r69_sanitize")
target_link_libraries(sanitize_r69 PRIVATE tiling_api register platform unified_dlog dl m graph_base)
target_compile_options(sanitize_r69 PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=dav-2201> $<$<COMPILE_LANGUAGE:ASC>:--cce-enable-sanitizer> $<$<COMPILE_LANGUAGE:ASC>:-gline-tables-only>)
add_executable(sanitize_r72 main.asc)
target_compile_definitions(sanitize_r72 PRIVATE BMMS_KERNEL_HEADER="r72_san_module.asc" BMMS_VARIANT="r72_sanitize_module")
target_link_libraries(sanitize_r72 PRIVATE tiling_api register platform unified_dlog dl m graph_base)
target_compile_options(sanitize_r72 PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=dav-2201> $<$<COMPILE_LANGUAGE:ASC>:--cce-enable-sanitizer> $<$<COMPILE_LANGUAGE:ASC>:-gline-tables-only>)
