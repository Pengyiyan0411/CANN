namespace bmms1269 {
struct Plan {int32_t B,M,N,K,tiny,workers;};
static inline bool Eligible(int B,int M,int N,int K){
    return B>=1&&B<=64&&M>=2&&M<=16&&N>=2&&N<=16&&K>=32&&K<=128&&K%8==0&&
        (bmmmaxsum_v43::UseTiny(M,N,K)||bmms71::UseResident(M,N,K));
}
template<class T,bool TA,bool TB,int K>
__aicore__ inline void Device(GM_ADDR a,GM_ADDR b,GM_ADDR y,const Plan& p){
    AscendC::TPipe pipe;
    constexpr int kp=K<=32?32:K<=64?64:128;
    constexpr bool DIRECT_A=!TA&&K==kp,DIRECT_B=TB&&K==kp;
    const int M=p.M,N=p.N,NP=(N+7)/8*8,D=M*NP;
    const bool directB=DIRECT_B&&N==NP;
    const int ar=(M*K+15)/16*16,br=(N*K+15)/16*16;
    const int af=(M*K+7)/8*8,bf=(N*K+7)/8*8;
    AscendC::TQue<AscendC::TPosition::VECIN,1> iq;
    AscendC::TQue<AscendC::TPosition::VECOUT,1> oq;
    AscendC::TBuf<AscendC::TPosition::VECCALC> floats,indices,packed,products,values,scratch;
    pipe.InitBuffer(iq,1,(ar+br)*sizeof(T));pipe.InitBuffer(oq,1,32);
    pipe.InitBuffer(floats,(af+bf)*4);pipe.InitBuffer(indices,2*kp*4);
    pipe.InitBuffer(packed,((DIRECT_A?0:M)+NP)*kp*4+32);
    pipe.InitBuffer(products,D*kp*4);pipe.InitBuffer(values,(2*D+M*8)*4);
    pipe.InitBuffer(scratch,64);
    auto fa=floats.Get<float>(),fb=fa[af],pack=packed.Get<float>();
    auto bp=directB?fb:pack,ap=DIRECT_A?fa:pack[NP*kp];
    auto prod=products.Get<float>(),dots=values.Get<float>(),tails=dots[D],rows=dots[2*D];
    auto tmp=scratch.Get<float>();
    auto ia=indices.Get<int32_t>(),ib=ia[kp];
    if constexpr(!DIRECT_A)AscendC::CreateVecIndex(ia,int32_t(0),K);
    if(!directB)AscendC::CreateVecIndex(ib,int32_t(0),K);
    AscendC::PipeBarrier<PIPE_V>();
    if constexpr(!DIRECT_A)AscendC::Muls(ia,ia,int32_t((TA?M:1)*4),K);
    if(!directB)AscendC::Muls(ib,ib,int32_t((TB?1:N)*4),K);
    AscendC::PipeBarrier<PIPE_V>();
    auto iau=ia.template ReinterpretCast<uint32_t>(),ibu=ib.template ReinterpretCast<uint32_t>();
    AscendC::GlobalTensor<T> ga,gb;AscendC::GlobalTensor<float> gy;
    ga.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(a),p.B*M*K);
    gb.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(b),p.B*N*K);
    gy.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(y),p.B);
    const uint8_t stride=static_cast<uint8_t>(kp/8);
    const AscendC::BinaryRepeatParams mulp{1,1,1,stride,0,stride};
    for(int batch=AscendC::GetBlockIdx();batch<p.B;batch+=p.workers){
        auto in=iq.AllocTensor<T>();AscendC::DataCopyPadExtParams<T> pd{false,0,0,T(0)};
        AscendC::DataCopyExtParams ac{1,uint32_t(M*K*sizeof(T)),0,0,0},bc{1,uint32_t(N*K*sizeof(T)),0,0,0};
        AscendC::DataCopyPad(in,ga[batch*M*K],ac,pd);AscendC::DataCopyPad(in[ar],gb[batch*N*K],bc,pd);
        iq.EnQue(in);in=iq.DeQue<T>();
        AscendC::Cast(fa,in,AscendC::RoundMode::CAST_NONE,M*K);
        AscendC::Cast(fb,in[ar],AscendC::RoundMode::CAST_NONE,N*K);
        if(!directB)AscendC::Duplicate(bp,0.0f,NP*kp);
        if constexpr(!DIRECT_A)AscendC::Duplicate(ap,0.0f,M*kp);
        AscendC::Duplicate(rows,0.0f,M*8);
        AscendC::PipeBarrier<PIPE_V>();
        if(!directB)for(int n=0;n<N;++n)AscendC::Gather(bp[n*kp],fb,ibu,uint32_t((TB?n*K:n)*4),K);
        if constexpr(!DIRECT_A)for(int m=0;m<M;++m)AscendC::Gather(ap[m*kp],fa,iau,uint32_t((TA?m:m*K)*4),K);
        AscendC::PipeBarrier<PIPE_V>();
        for(int m=0;m<M;++m)for(int k=0;k<kp;k+=64)
            AscendC::Mul(prod[m*NP*kp+k],ap[m*kp+k],bp[k],bmms83::MinI(kp,64),NP,mulp);
        AscendC::PipeBarrier<PIPE_V>();
        if(p.tiny){
            // Reduce every dot together. Split at 248 to keep every destination
            // 32-byte aligned while respecting the 255-repeat instruction limit.
            for(int d=0;d<D;d+=248){
                const int reps=bmms83::MinI(248,D-d);
                AscendC::WholeReduceSum(dots[d],prod[d*kp],bmms83::MinI(K,64),reps,1,1,kp/8);
                if constexpr(K>64)AscendC::WholeReduceSum(tails[d],prod[d*kp+64],K-64,reps,1,1,kp/8);
            }
            if constexpr(K>64){AscendC::PipeBarrier<PIPE_V>();AscendC::Add(dots,dots,tails,D);}
        }else{
            if constexpr(kp==128){
                const AscendC::BinaryRepeatParams addp{1,1,1,16,16,16};
                for(int d=0;d<D;d+=248)
                    AscendC::Add(prod[d*kp],prod[d*kp],prod[d*kp+64],64,bmms83::MinI(248,D-d),addp);
                AscendC::PipeBarrier<PIPE_V>();
            }
            for(int d=0;d<D;d+=248)
                AscendC::WholeReduceSum(dots[d],prod[d*kp],bmms83::MinI(kp,64),bmms83::MinI(248,D-d),1,1,kp/8);
        }
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::WholeReduceMax(rows,dots,N,M,8,1,NP/8,AscendC::ReduceOrder::ORDER_ONLY_VALUE);
        AscendC::PipeBarrier<PIPE_V>();
        auto out=oq.AllocTensor<float>();
        if(M<=8){AscendC::WholeReduceSum(out,rows,M*8,1,1,1,8);}
        else{
            AscendC::WholeReduceSum(tmp,rows,64,1,1,1,8);
            AscendC::WholeReduceSum(tmp[8],rows[64],M*8-64,1,1,1,8);
            AscendC::PipeBarrier<PIPE_V>();AscendC::Add(out,tmp,tmp[8],1);
        }
        oq.EnQue(out);out=oq.DeQue<float>();AscendC::DataCopyExtParams cp{1,4,0,0,0};
        AscendC::DataCopyPad(gy[batch],out,cp);oq.FreeTensor(out);iq.FreeTensor(in);
    }
}
} // namespace bmms1269
